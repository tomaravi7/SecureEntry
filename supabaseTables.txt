-- Create enum for notification types
CREATE TYPE notification_type AS ENUM (
    'expected_guest',
    'expected_package',
    'file_complaint',
    'emergency'
);

-- Create enum for notification status
CREATE TYPE notification_status AS ENUM (
    'pending',
    'resolved'
);

-- Create notifications table
CREATE TABLE notifications (
    id BIGSERIAL PRIMARY KEY,
    resident_id UUID NOT NULL,
    type notification_type NOT NULL,
    status notification_status NOT NULL DEFAULT 'pending',
    timestamp TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    details TEXT,
    FOREIGN KEY (resident_id) REFERENCES auth.users(id)
);

-- Create index on status and timestamp for efficient queries
CREATE INDEX idx_notifications_status_timestamp ON notifications(status, timestamp);

-- Create index on resident_id for efficient queries
CREATE INDEX idx_notifications_resident_id ON notifications(resident_id);

-- Grant access to authenticated users
ALTER TABLE notifications ENABLE ROW LEVEL SECURITY;

-- Policy to allow residents to insert their own notifications
CREATE POLICY insert_own_notifications ON notifications
    FOR INSERT TO authenticated
    WITH CHECK (auth.uid() = resident_id);

-- Policy to allow residents to view their own notifications
CREATE POLICY select_own_notifications ON notifications
    FOR SELECT TO authenticated
    USING (auth.uid() = resident_id);

-- Policy to allow guards to view all pending notifications
CREATE POLICY select_pending_notifications ON notifications
    FOR SELECT TO authenticated
    USING (status = 'pending' AND auth.jwt() ->> 'role' = 'guard');

-- Policy to allow guards to update notification status
CREATE POLICY update_notification_status ON notifications
    FOR UPDATE TO authenticated
    USING (auth.jwt() ->> 'role' = 'guard')
    WITH CHECK (status = 'resolved');

);


-- Create the user_roles table
CREATE TABLE user_roles (
  id UUID REFERENCES auth.users(id) PRIMARY KEY,
  role TEXT NOT NULL CHECK (role IN ('admin', 'resident', 'guard')),
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- Create a function to automatically update the updated_at column
CREATE OR REPLACE FUNCTION update_updated_at_column()
RETURNS TRIGGER AS $$
BEGIN
   NEW.updated_at = NOW();
   RETURN NEW;
END;
$$ language 'plpgsql';

-- Create a trigger to automatically update the updated_at column
CREATE TRIGGER update_user_roles_updated_at
BEFORE UPDATE ON user_roles
FOR EACH ROW
EXECUTE FUNCTION update_updated_at_column();
